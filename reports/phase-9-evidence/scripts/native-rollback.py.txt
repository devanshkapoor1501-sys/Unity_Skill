import json, hashlib
from pathlib import Path
from importlib.resources import files
from dataclasses import asdict
from polyforge.contracts.validation import load_json
from polyforge.runner.processes import run_process,ProcessPolicy
from polyforge.runner.locks import unity_project_lock
from polyforge.runner.state import atomic_json
from polyforge.security.paths import WorkspacePolicy
from polyforge.unity.bridge import configuration, unity_tool,launch
from polyforge.unity.upgrade import sync_package
from polyforge.unity import recovery
from polyforge.unity.evidence import reconcile_report

p=WorkspacePolicy(Path('.tmp/phase6/arena'));c=configuration(p);exe=unity_tool(c)
out=Path('.tmp/phase9'); identifier='phase9_cross_tool'
resources=Path(str(files('unity_package')))
desired={f.relative_to(resources).as_posix():f.read_bytes() for f in resources.rglob('*') if f.is_file() and (f.suffix in {'.cs','.asmdef'} or f.name=='package.json')}
desired['Editor/PolyForgeRecoveryFaultFixture.cs']=Path('tests/unity_recovery_fault.cs').read_bytes()
with unity_project_lock(p,p.path('UnityProject')):
    print('Owned fixture package sync',sync_package(p,desired),flush=True)

actual_launch=recovery.launch
def log_launch(*args,**kwargs):
    print('Native recovery stage:',args[3],args[4],flush=True)
    return actual_launch(*args,**kwargs)
recovery.launch=log_launch
print('Capture full native prestate and snapshot',flush=True)
tx=recovery.prepare_transaction(p,identifier)
atomic_json(p,tx.relative+'/qualification-intent.json',{'faults':['registered Blender source corruption','manifest seed corruption','importer scale/cache drift','prefab scale','missing source binding','missing material reference'],'original':tx.read()['original']})
print('Prepared',identifier,'files',len(tx.read()['original']),'dependencies',len(tx.read()['dependencies']),flush=True)
with unity_project_lock(p,p.path('UnityProject')):
    relative='runs/phase9-native-fault'
    args=[str(exe),'-batchmode','-nographics','-job-worker-count','2','-projectPath',str(p.path('UnityProject')),'-logFile','-','-upmLogFile',str(p.path(relative+'/upm.log')),'-executeMethod','PolyForgeRecoveryFaultFixture.Inject']
    outcome=run_process(args,p,relative,ProcessPolicy((exe,),c.unity_timeout_seconds,c.max_output_bytes,c.memory_limit_mb))
    atomic_json(p,relative+'/process.json',{'argv':args,**asdict(outcome)})
    assert outcome.returncode==0 and outcome.reason=='exit',asdict(outcome)
    observed=p.path(relative+'/bindings.json')
    launch(p,exe,c,relative+'/diagnose','reconcile',request=p.path('.polyforge/unity-request.json'),output=observed,allow_contract_failure=True)
    diagnosis=reconcile_report(load_json(p.path('.polyforge/unity-request.json')),load_json(observed))
    assert diagnosis['status']=='FAIL',diagnosis
    atomic_json(p,relative+'/diagnosis.json',diagnosis)
    print('Native injected issue categories',sorted({i['category'] for i in diagnosis['issues']}),flush=True)
    blend=next(n for n in tx.read()['original'] if n.endswith('.blend'))
    p.path(blend).write_bytes(b'controlled corrupted registered Blender source')
    manifest=load_json(p.path('game_manifest.json'));manifest['project']['seed']=43
    atomic_json(p,'game_manifest.json',manifest)
    tx.record('APPLIED',qualification_faults=True,changed_source=blend,source_before_sha256=tx.read()['original'][blend],source_after_sha256=hashlib.sha256(p.path(blend).read_bytes()).hexdigest())
print('Restore source bytes, force native reimport and replay all applicable regressions',flush=True)
result=recovery.repair_rollback(p.root,identifier)
(out/'native-rollback.json').write_text(json.dumps(result.to_dict(),indent=2)+'\n',encoding='utf-8')
print('Native rollback',result.status,result.error_code,flush=True)
raise SystemExit(result.exit_code)
