import json,hashlib
from pathlib import Path
from polyforge.security.paths import WorkspacePolicy
from polyforge.unity import recovery
from polyforge.unity.transactions import ManagedTransaction

p=WorkspacePolicy(Path('.tmp/phase6/arena'));identifier='phase9_crash_recovery'
proof=json.loads(p.path('runs/phase9-final-timeout-proof.json').read_text());prior=proof['attempt']
before={n:hashlib.sha256(p.path(prior+'/native/'+n).read_bytes()).hexdigest() for n in ['process.json','stdout.txt','stderr.txt']}
actual=recovery.launch
def log(*args,**kwargs):
 print('Native stage',args[3],args[4],flush=True)
 return actual(*args,**kwargs)
recovery.launch=log
result=recovery.resume_run(p.root,identifier)
Path('.tmp/phase9/native-resume-final.json').write_text(json.dumps(result.to_dict(),indent=2)+'\n')
assert result.status=='PASS',result.to_dict()
assert ManagedTransaction(p,identifier).verify_restored()
assert not p.path('UnityProject/.polyforge/repair-quarantine.json').exists()
after={n:hashlib.sha256(p.path(prior+'/native/'+n).read_bytes()).hexdigest() for n in before}
assert before==after
assert ManagedTransaction(p,identifier).read()['recovery_attempt']!=prior
Path('.tmp/phase9/retained-timeout-check.json').write_text(json.dumps({'status':'PASS','failed_attempt':prior,'failed_process':proof['process'],'before':before,'after':after},indent=2)+'\n')
print('Native resume, complete invariant replay and retained negative evidence PASS',flush=True)
