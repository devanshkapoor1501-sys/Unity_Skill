"""Shared-style preservation and actual registered-source/native rollback proof."""
import hashlib
import json
import shutil
import sys
from dataclasses import asdict
from importlib.resources import files
from pathlib import Path

from polyforge.capabilities.protocol import RunContext
from polyforge.contracts.validation import load_json
from polyforge.runner.locks import unity_project_lock
from polyforge.runner.processes import ProcessPolicy, run_process
from polyforge.runner.state import atomic_json
from polyforge.security.paths import WorkspacePolicy
from polyforge.sprites.integration import import_strip
from polyforge.unity import recovery
from polyforge.unity.bridge import configuration, unity_tool
from polyforge.unity.platformer import build_platformer
from polyforge.unity.upgrade import sync_package

base = Path('.tmp/phase11').resolve()
retry = '--retry' in sys.argv
policy = WorkspacePolicy(base / ('demo' if retry else 'arena-final'))
prefix = 'shared-retry' if retry else 'shared'
source = Path('.tmp/phase8/direct-final').resolve()
if not policy.path('input').exists():
    shutil.copytree(source / 'input', policy.path('input'))
else:
    for path in (source / 'input').rglob('*'):
        if path.is_file():
            assert policy.path(Path('input') / path.relative_to(source / 'input')).read_bytes() == path.read_bytes()
plan = load_json(source / 'workforce/task_27bd6f684c81a83a_b15bbf5c/plan.json')
print('Import approved sprite into the actual original arena project', flush=True)
sprites = import_strip(plan['original_request']['input'], RunContext(policy.root))
(base / (prefix + '-sprites-result.json')).write_text(json.dumps(sprites.to_dict(), indent=2), encoding='utf-8')
assert sprites.status == 'PASS', sprites.to_dict()
request = {'schema_version': '1.0.0', 'scene_id': 'shared_checkpoint_course', 'seed': 42,
           'sprite_receipt': f'runs/{sprites.run_id}/bindings.json', 'layout': 'checkpoint_course_v1'}
print('Build distinct checkpoint style, preserving all existing 3D generated bytes', flush=True)
platformer = build_platformer(policy.root, request)
(base / (prefix + '-platformer-result.json')).write_text(json.dumps(platformer.to_dict(), indent=2), encoding='utf-8')
assert platformer.status == 'PASS', platformer.to_dict()
resources = Path(str(files('unity_package')))
desired = {p.relative_to(resources).as_posix(): p.read_bytes()
           for p in resources.rglob('*') if p.is_file() and (p.suffix in {'.cs', '.asmdef'} or p.name == 'package.json')}
desired['Editor/PolyForgeRecoveryFaultFixture.cs'] = Path('tests/unity_recovery_fault.cs').read_bytes()
with unity_project_lock(policy, policy.path('UnityProject')):
    sync_package(policy, desired)
print('Capture native prestate and every applicable game/sprite regression', flush=True)
tx = recovery.prepare_transaction(policy, 'phase11_shared_cross_tool' + ('_retry' if retry else ''))
config = configuration(policy)
executable = unity_tool(config)
with unity_project_lock(policy, policy.path('UnityProject')):
    relative = 'runs/phase11-native-rollback-fault'
    argv = [str(executable), '-batchmode', '-nographics', '-job-worker-count', '2',
            '-projectPath', str(policy.path('UnityProject')), '-logFile', '-',
            '-upmLogFile', str(policy.path(relative + '/upm.log')),
            '-executeMethod', 'PolyForgeRecoveryFaultFixture.Inject']
    outcome = run_process(argv, policy, relative, ProcessPolicy(
        (executable,), config.unity_timeout_seconds, config.max_output_bytes, config.memory_limit_mb))
    atomic_json(policy, relative + '/process.json', {'argv': argv, **asdict(outcome)})
    assert outcome.reason == 'exit' and outcome.returncode == 0, asdict(outcome)
    blend = next(name for name in tx.read()['original'] if name.endswith('.blend'))
    policy.path(blend).write_bytes(b'controlled registered source corruption')
    manifest = load_json(policy.path('game_manifest.json'))
    manifest['project']['seed'] = 43
    atomic_json(policy, 'game_manifest.json', manifest)
    tx.record('APPLIED', qualification_faults=True, changed_source=blend,
              source_after_sha256=hashlib.sha256(policy.path(blend).read_bytes()).hexdigest())
print('Restore snapshot, reimport natively and replay all original and new-style oracles', flush=True)
restored = recovery.repair_rollback(policy.root, tx.identifier)
(base / (prefix + '-native-rollback-result.json')).write_text(json.dumps(restored.to_dict(), indent=2), encoding='utf-8')
print(json.dumps(restored.to_dict(), indent=2), flush=True)
assert restored.status == 'PASS', restored.to_dict()
