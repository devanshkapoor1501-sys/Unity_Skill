import hashlib,json,re,sys,xml.etree.ElementTree as ET

from polyforge.knowledge.probes import read_probe

install=json.loads((base/'install/commands.json').read_text());assert len(install)==14 and all(c['exit_code']==c['expected_exit'] for c in install)
implementation={
