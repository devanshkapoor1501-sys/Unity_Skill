from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
import json,time
from polyforge.assets.pipeline import build_asset
from polyforge.capabilities.protocol import RunContext
p=Path('.tmp/phase8/assets');inputs=json.loads(Path('.tmp/phase8/asset-inputs.json').read_text());started={};ended={}
def run(v):
 started[v['asset_id']]=time.monotonic()
 r=build_asset({'schema_version':'1.0.0','spec':v},RunContext(p));ended[v['asset_id']]=time.monotonic()
 assert r.status=='PASS',r.to_dict()
 return r.to_dict()
with ThreadPoolExecutor(max_workers=2) as pool:results=list(pool.map(run,inputs))
assert max(started.values())<min(ended.values())
Path('.tmp/phase8/asset-results.json').write_text(json.dumps({'results':results,'started':started,'ended':ended,'overlapping_jobs':True,'native_blender_parallelism':1},indent=2)+'\n')
print('PASS two asset types, overlapping workflows, distinct source paths',flush=True)
